\section*{S2. Inference safeguards}
A submission that fails or exceeds the 9-h limit yields no score, so the student stage was written to fall back to the
parent's predictions. Algorithm~\ref{alg:fusion} is the procedure used by the final system; it changes nothing in a run
without failures.

\begin{algorithm}[h]
\caption{Rank fusion with fall-back to the parent predictions}
\label{alg:fusion}
\begin{algorithmic}[1]
\Require parent predictions $P$ (file \texttt{sub}); student checkpoints with hashes $\{(\theta_m,h_m)\}_{m=1}^{M}$;
time limit $T_{\max}$; elapsed time $T_0$; weight $w$
\Ensure final predictions $F$
\State $P_0\gets$ byte-exact copy of \texttt{sub}
\State verify $\operatorname{SHA256}(\theta_m)=h_m$ for all $m$; on mismatch \Return $P_0$
\State run the students with timeout $T_{\max}-T_0-20\,\text{min}$; on error or timeout \Return $P_0$
\State $\bar S\gets\frac{1}{M}\sum_m\sigma(f_{\theta_m}(X))$
\For{$c=1,\dots,12$}
  \State $\rho^P\gets r(P_{\cdot c})$, $\rho^S\gets r(\bar S_{\cdot c})$
  \State $\mathcal{U}\gets\{i:\bar S_{ic}\text{ missing}\}$; fit $\rho^S\approx\alpha+\beta\rho^P$ on $i\notin\mathcal{U}$; impute $\rho^S_i$ for $i\in\mathcal{U}$
  \State $F_{\cdot c}\gets(1-w)\rho^P+w\rho^S$
\EndFor
\If{$F$ fails the schema, row-count or finiteness checks} \Return $P_0$ \EndIf
\State \Return $F$
\end{algorithmic}
\end{algorithm}
